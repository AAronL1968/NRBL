Public Sub sub_run_all_imports()
    Forms!frm_imports!btn_run_all_imports.BackColor = CLR_BRIGHT_YELLOW
    Forms!frm_imports.Requery
    Forms!frm_imports.Repaint
    Dim strStatSource As String
    Dim strHitterButton As String
    Dim strPitcherButton As String
    Dim iEnabled As Long
    Dim iFanGraphs As Long
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Dim ctlLoop As Control
    Dim ctlAction As Control
    Dim strControlName As String
    Dim strExecute As String
    
    Set db = CurrentDb
    Set rst = db.OpenRecordset("qry_external_stat_sources")
    Do While Not rst.EOF
        strStatSource = IIf(IsNull(rst!stat_source), "", rst!stat_source)
        iFanGraphs = IIf(IsNull(rst!fangraphs), 0, Abs(CLng(rst!fangraphs)))
        iEnabled = IIf(IsNull(rst!Enabled), 0, Abs(CLng(rst!Enabled)))
        strHitterButton = IIf(IsNull(rst!btn_name_import_hitters), "", rst!btn_name_import_hitters)
        strPitcherButton = IIf(IsNull(rst!btn_name_import_pitchers), "", rst!btn_name_import_pitchers)
        'Debug.Print strStatSource
        For Each ctlLoop In Forms!frm_imports.Controls
            strControlName = ctlLoop.Name
            If strControlName = strHitterButton Or strControlName = strPitcherButton Then
                If iEnabled = 0 Then
                    ctlLoop.Enabled = False
                Else
                    ctlLoop.Enabled = True
                    ctlLoop.BackColor = CLR_BLUE_QUEUED
                End If
            End If
        Next
        rst.MoveNext
    Loop
    rst.Close
    Set rst = Nothing
    db.Close
    Set db = Nothing
        
    Forms!frm_imports.Requery
    Forms!frm_imports.Repaint
    Call fun_import_onroto_hitters
    Call fun_set_sources
    Call fun_import_onroto_pitchers
    Call fun_set_sources
    Call fun_load_frm
    
    Set db = CurrentDb
    Set rst = db.OpenRecordset("qry_external_stat_sources")
    Do While Not rst.EOF
        strStatSource = IIf(IsNull(rst!stat_source), "", rst!stat_source)
        iFanGraphs = IIf(IsNull(rst!fangraphs), 0, Abs(CLng(rst!fangraphs)))
        iEnabled = IIf(IsNull(rst!Enabled), 0, Abs(CLng(rst!Enabled)))
        strHitterButton = IIf(IsNull(rst!btn_name_import_hitters), "", rst!btn_name_import_hitters)
        strPitcherButton = IIf(IsNull(rst!btn_name_import_pitchers), "", rst!btn_name_import_pitchers)
        If iFanGraphs <> 0 And iEnabled <> 0 Then
            If strHitterButton <> "" Then
                Call fun_import_fangraphs_hitters(strStatSource, False)
            End If
            If strPitcherButton <> "" Then
                Call fun_import_fangraphs_pitchers(strStatSource, False)
            End If
            Call fun_set_sources
            Call fun_load_frm
        End If
        rst.MoveNext
    Loop
    rst.Close
    Set rst = Nothing
    db.Close
    Set db = Nothing
    
    DoCmd.OpenQuery "import_players_10_update_first_and_last_names"
    Forms!frm_imports!btn_run_all_imports.BackColor = CLR_GREEN_BTN
    Forms!frm_imports.Requery
    Forms!frm_imports.Repaint
End Sub
Public Function fun_import_fangraphs_pitchers(strStatSource As String, boolTreatDrafteesAsKeepers As Boolean)

    '
    ' Get info (source csv linked table, abbreviation) for specified stat source from the [external_stat_sources] table
    '
    strStatus = "strStatSource = " & strStatSource & ", truncate fg_selected_external_stat_source..."
    Call fun_upd_import_status(strStatus)
    strSQL = ""
    strSQL = strSQL & "delete fg_selected_external_stat_source.stat_source"
    strSQL = strSQL & "  from fg_selected_external_stat_source;"
'Debug.Print strSQL
    db.Execute strSQL, dbFailOnError
    strStatus = "Populate fg_selected_external_stat_source with " & strStatSource & " data..."
    Call fun_upd_import_status(strStatus)
    strSQL = ""
    strSQL = strSQL & "insert into fg_selected_external_stat_source ( "
    strSQL = strSQL & "       stat_source, preference_order, csv_source_table_hitters, csv_source_table_pitchers, "
    strSQL = strSQL & "       stat_source_abbreviation, fangraphs, enabled )"
    strSQL = strSQL & "select external_stat_sources.stat_source, external_stat_sources.preference_order, "
    strSQL = strSQL & "       external_stat_sources.csv_source_table_hitters, external_stat_sources.csv_source_table_pitchers, "
    strSQL = strSQL & "       external_stat_sources.stat_source_abbreviation , external_stat_sources.fangraphs, external_stat_sources.enabled "
    strSQL = strSQL & "  from external_stat_sources "
    strSQL = strSQL & " where external_stat_sources.stat_source = " & """" & strStatSource & """" & ";"
'Debug.Print strSQL
    db.Execute strSQL, dbFailOnError
    Set rst = db.OpenRecordset("fg_selected_external_stat_source")
    strAbbrev = IIf(IsNull(rst!stat_source_abbreviation), "", rst!stat_source_abbreviation)
    strSourceTable = IIf(IsNull(rst!csv_source_table_pitchers), "", rst!csv_source_table_pitchers)
    rst.Close
    strPitchingProjectionsTableName = "fg_pitching_projections_" & strAbbrev
    strStatsCopyTableName = "fg_stats_copy_pitchers_" & strAbbrev
    If boolTreatDrafteesAsKeepers = True Then
        strStatsCopyTableName = "fg_stats_copy_pitchers_" & strAbbrev & "_in_game"
    End If
    strButtonName = "btn_import_" & strAbbrev & "_pitchers"
    If CurrentProject.AllForms("frm_imports").IsLoaded Then
        For Each ctl In Forms!frm_imports.Controls
            ' Turn the button yellow
            If ctl.Name = strButtonName Then
                ctl.BackColor = CLR_BRIGHT_YELLOW
            End If
        Next ctl
        'MsgBox "strAbbrev: " & strAbbrev & vbCrLf & "strSourceTable: " & strSourceTable
        Forms!frm_imports.Requery
        Forms!frm_imports.Repaint
    End If
    Call fun_import_fangraphs_pitchers_build_objects(strStatSource, boolTreatDrafteesAsKeepers)
    Call fun_import_fangraphs_pitchers_populate_objects(strStatSource, boolTreatDrafteesAsKeepers)
    Call fun_import_fangraphs_pitchers_calculate_values(strStatSource, boolTreatDrafteesAsKeepers)
    
    '
    ' Finish up!
    '
    'Debug.Print "strAbbrev: " & strAbbrev
    'Debug.Print "strSourceTable: " & strSourceTable
    'Debug.Print "strPitchingProjectionsTableName: " & strPitchingProjectionsTableName
    'Debug.Print "strStatsCopyTableName: " & strStatsCopyTableName
    'Debug.Print "fPitcherPool: " & fPitcherPool
    'Debug.Print "strButtonName: " & strButtonName
    Call fun_load_stat_copy_aggs
    If CurrentProject.AllForms("frm_imports").IsLoaded Then
        For Each ctl In Forms!frm_imports.Controls
            If ctl.Name = strButtonName Then
                ' Turn the button green
                ctl.BackColor = CLR_GREEN_BTN
            End If
        Next ctl
        Forms!frm_imports.Requery
    End If
    Call fun_upd_import_status("Done (" & Now() & ").  Ready.")
    'MsgBox "strAbbrev: " & strAbbrev & vbCrLf & "strSourceTable: " & strSourceTable
    Set rst = Nothing
    Set db = Nothing
End Function

Public Function fun_import_fangraphs_pitchers_populate_objects(strStatSource As String, boolTreatDrafteesAsKeepers As Boolean)
    DoCmd.SetWarnings True
    Dim strTableName As String
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Dim strSQL As String
    Dim strStatus As String
    Dim strAbbrev As String
    Dim strSourceTable As String
    Dim strPitchingProjectionsTableName As String
    Dim strStatsCopyTableName As String
    Set db = CurrentDb
    Set rst = db.OpenRecordset("fg_selected_external_stat_source")
    strAbbrev = IIf(IsNull(rst!stat_source_abbreviation), "", rst!stat_source_abbreviation)
    strSourceTable = IIf(IsNull(rst!csv_source_table_pitchers), "", rst!csv_source_table_pitchers)
    rst.Close
    Set rst = Nothing
    strPitchingProjectionsTableName = "fg_pitching_projections_" & strAbbrev
    strStatsCopyTableName = "fg_stats_copy_pitchers_" & strAbbrev
    If boolTreatDrafteesAsKeepers = True Then
        strStatsCopyTableName = "fg_stats_copy_pitchers_" & strAbbrev & "_in_game"
    End If
    If boolTreatDrafteesAsKeepers = False Then ' Populate tables for full import
        '
        ' Populate pitching projections table from csv
        '
        strStatus = "Populate " & strPitchingProjectionsTableName & " from " & strSourceTable & "..."
        Call fun_upd_import_status(strStatus)
        strSQL = ""
        strSQL = strSQL & "insert into [" & strPitchingProjectionsTableName & "] ( stat_source, Name, Team, W, L, QS, ERA, G, GS, SV, HLD, BS, IP, TBF, H, R, ER, HR, BB, IBB, HBP, SO, [K/9],"
        strSQL = strSQL & " [BB/9], [K/BB], [HR/9], [K%], [BB%], [K-BB%], [AVG], WHIP, BABIP, [LOB%], [GB%], [HR/FB], FIP, WAR, [RA9-WAR], ADP, InterSD, InterSK, IntraSD,"
        strSQL = strSQL & " Vol, Skew, Dim, FPTS, [FPTS/IP], SPTS, [SPTS/IP], NameASCII, PlayerId, MLBAMID ) select DISTINCT " & """" & strStatSource & """" & " as stat_source, Name, Team, W, L, QS, "
        strSQL = strSQL & " ERA, G, GS, SV, HLD, BS, IP, TBF, H, R, ER, HR, BB, IBB, HBP, SO, [K/9], [BB/9], [K/BB], [HR/9], [K%], [BB%], [K-BB%], Avg, WHIP, BABIP, [LOB%], [GB%], [HR/FB], FIP, WAR,"
        strSQL = strSQL & " [RA9-WAR], ADP, InterSD, InterSK, IntraSD, Vol, Skew, Dim, FPTS, [FPTS/IP], SPTS, [SPTS/IP], NameASCII, PlayerId, MLBAMID"
        strSQL = strSQL & " from [" & strSourceTable & "] WHERE [G] > 1;"
        db.Execute strSQL, dbFailOnError
        '
        ' Clean/normalize team names
        '
        strStatus = "Clean NL team names in " & strPitchingProjectionsTableName & "..."
        Call fun_upd_import_status(strStatus)
        strSQL = ""
        strSQL = strSQL & "update MLB_team_aliases inner join [" & strPitchingProjectionsTableName & "] "
        strSQL = strSQL & "on MLB_team_aliases.Alias = [" & strPitchingProjectionsTableName & "].Team "
        strSQL = strSQL & "set [" & strPitchingProjectionsTableName & "].Team = MLB_team_aliases.Team;"
    'Debug.Print strSQL
        db.Execute strSQL, dbFailOnError
        '
        ' Set OnRoto ID
        '
        strStatus = "Set OnRoto ID (pass #1) in " & strPitchingProjectionsTableName & "..."
        Call fun_upd_import_status(strStatus)
        strSQL = ""
        strSQL = strSQL & "update NRBL_players inner join [" & strPitchingProjectionsTableName & "] on (NRBL_players.steamer_id = [" & strPitchingProjectionsTableName & "].PlayerId) "

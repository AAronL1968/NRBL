Public Sub search_querydef_sql()
    Dim db As DAO.Database
    Dim qdf As DAO.QueryDef
    Dim queryName As String
    Dim querySQL As String
    Set db = CurrentDb()
    For Each qdf In db.QueryDefs
        queryName = qdf.Name
        querySQL = qdf.sql
        If InStr(1, querySQL, "AB") > 0 Then
            MsgBox queryName & " | " & querySQL
        End If
    Next qdf
    MsgBox "Done."
End Sub
Public Function fun_hitter_pool() As Currency
    If Not CurrentProject.AllForms("frm_imports").IsLoaded Then
        DoCmd.OpenForm "frm_imports"
    End If
    Dim fPoolAmount As Currency
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Dim fDraftCap As Currency
    Dim iTeamCt As Integer
    Dim iHitterSlots As Integer
    Dim iPitcherSlots As Integer
    Dim iSwingmanSlots As Integer
    Dim iValidSlots As Integer
    Dim fHitterPct As Double
    Dim fPitcherPct As Double
    Set db = CurrentDb()
    Set rst = db.OpenRecordset("NRBL_application_settings")
    fDraftCap = rst!draft_cap.Value
    iHitterSlots = rst!hitter_slots.Value
    iPitcherSlots = rst!pitcher_slots.Value
    iSwingmanSlots = rst!swingman_slots.Value
    fHitterPct = rst!league_hitter_salary_pct.Value
    fPitcherPct = rst!league_pitcher_salary_pct.Value
    iTeamCt = rst!active_team_count.Value
    rst.Close
    Set rst = Nothing
    Set db = Nothing
    iValidSlots = iHitterSlots + iSwingmanSlots
    fPoolAmount = (fDraftCap * iValidSlots * fHitterPct) - (iTeamCt * iValidSlots)  ' 65% (or whatever hitter percentage value is set) of total money available, minus $1 min bid per hitter
    
    Forms!frm_imports!txt_team_ct.Value = iTeamCt
    Forms!frm_imports!txt_draft_cap.Value = fDraftCap
    Forms!frm_imports!txt_hitter_slots.Value = iHitterSlots
    Forms!frm_imports!txt_pitcher_slots.Value = iPitcherSlots
    Forms!frm_imports!txt_swingman_slots.Value = iSwingmanSlots
    Forms!frm_imports!txt_hitter_pct.Value = fHitterPct
    Forms!frm_imports!txt_pitcher_pct.Value = fPitcherPct
    fun_hitter_pool = fPoolAmount
End Function

Public Function fun_set_status()

    Dim iCNeeded As Integer

    Dim i2BNeeded As Integer
    Dim iSSNeeded As Integer
    Dim i3BNeeded As Integer
    Dim iMINeeded As Integer
    Dim iCMNeeded As Integer
    Dim iCOFNeeded As Integer
    Dim iUTNeeded As Integer
    Dim iSWNeeded As Integer
    Dim iPlayersNeeded As Integer
    Dim iNeedsComma As Integer
    Dim iMaxBid As Integer
    Dim iDraftPickCt As Integer
    Dim strLastName As String
    Dim strFirstName As String
    Dim strOpeningBidder As String
    Dim strWinningBidder As String
    Dim iOpeningBid As Integer
    Dim iSal As Integer
    Dim strNL As String
    Dim strPos As String
    Dim strLastDraftPick As String
    Dim iCt As Integer
    strLastDraftPick = ""
    Set db = CurrentDb()
    strOwner = IIf(IsNull(Forms![frm]!txt_nominator.Value), "", Forms![frm]!txt_nominator.Value)
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_delete"
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_append"
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_ct_delete"
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_ct_append"
    Set rst = db.OpenRecordset("tbl_team_needs_summary_nominator_ct")
    iCt = rst!ct
    rst.Close
    Set rst = Nothing
    iPsNeeded = 0
    iOFsNeeded = 0
    iCNeeded = 0
    i1BNeeded = 0
    i2BNeeded = 0
    iSSNeeded = 0
    i3BNeeded = 0
    iMINeeded = 0
    iCMNeeded = 0
    iCOFNeeded = 0
    iUTNeeded = 0
    iSWNeeded = 0
    iPlayersNeeded = 0
    iNeedsComma = 0
    If iCt = 0 Then
        strStatus = "The auction draft has completed."
    End If
    If iCt > 0 Then
        Set rst = db.OpenRecordset("tbl_team_needs_summary_nominator")
        iPsNeeded = IIf(IsNull(rst!Ps_needed), 0, rst!Ps_needed)
        iOFsNeeded = IIf(IsNull(rst!OFs_needed), 0, rst!OFs_needed)
        iCNeeded = IIf(IsNull(rst!needs_C), 0, rst!needs_C)
        i1BNeeded = IIf(IsNull(rst!needs_1B), 0, rst!needs_1B)
        i2BNeeded = IIf(IsNull(rst!needs_2B), 0, rst!needs_2B)
        iSSNeeded = IIf(IsNull(rst!needs_SS), 0, rst!needs_SS)
        i3BNeeded = IIf(IsNull(rst!needs_3B), 0, rst!needs_3B)
        iMINeeded = IIf(IsNull(rst!needs_MI), 0, rst!needs_MI)
        iCMNeeded = IIf(IsNull(rst!needs_CM), 0, rst!needs_CM)
        iCOFNeeded = IIf(IsNull(rst!needs_COF), 0, rst!needs_COF)
        iUTNeeded = IIf(IsNull(rst!needs_UT), 0, rst!needs_UT)
        iSWNeeded = IIf(IsNull(rst!needs_SW), 0, rst!needs_SW)
        iPlayersNeeded = IIf(IsNull(rst!players_needed), 0, rst!players_needed)
        iNeedsComma = 0
        rst.Close
        Set rst = db.OpenRecordset("tbl_max_bid")
        iMaxBid = IIf(IsNull(Forms!frm!txt_max_bid.Value), 0, Forms!frm!txt_max_bid.Value)
        If iMaxBid = 0 Then
            iMaxBid = IIf(IsNull(rst!max_bid_amt), 0, rst!max_bid_amt)
        End If
        rst.Close
        Set rst = Nothing
        DoCmd.OpenQuery "qry_draft_pick_ct_delete"
        DoCmd.OpenQuery "qry_draft_pick_ct_append"
    End If
    DoCmd.OpenQuery "qry_most_recent_drafted_player_delete"
    DoCmd.OpenQuery "qry_most_recent_drafted_player_append"
    Set rst = db.OpenRecordset("tbl_draft_pick_ct")
    iDraftPickCt = rst!draft_pick_ct
    rst.Close
    Set rst = Nothing
    If iDraftPickCt > 0 Then
        Set rst = db.OpenRecordset("tbl_most_recent_drafted_player")

        strFirstName = rst!first_name
        strOpeningBidder = rst!opening_bidder
        strWinningBidder = rst!Owner

        iSal = rst!Sal

        strPos = rst!Pos

    End If

                strStatus = strStatus & iOFsNeeded & " OF"
            Else
                strStatus = strStatus & "OF"
            End If
            iNeedsComma = 1
        End If
        If iCNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "C"
            iNeedsComma = 1
        End If
        If i1BNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "1B"
            iNeedsComma = 1
        End If
        If i2BNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "2B"
            iNeedsComma = 1
        End If
        If iSSNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "SS"
            iNeedsComma = 1
        End If
        If i3BNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "3B"
            iNeedsComma = 1
        End If
        If iMINeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "MI"
            iNeedsComma = 1
        End If
        If iCMNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "CM"
            iNeedsComma = 1

        If iCOFNeeded > 0 Then
            If iNeedsComma = 1 Then

            iNeedsComma = 1

            strStatus = strStatus & "UT"
            iNeedsComma = 1

            strStatus = strStatus & "SW"

Public Function fun_draft()
    Dim strPositionCategory As String
    Dim iOpeningBid As Integer
    Dim iWinningBid As Integer
    Dim iMaxBid As Integer
    Dim strBidder As String
    Dim iPlayerId As Long
    Dim iMinPosSlot As Long
    Dim iBid As Long
    Dim iInitialBid As Long
    Dim strNominator As String
    
    iBid = IIf(IsNull([Forms]![frm]![txt_bid].Value), 0, [Forms]![frm]![txt_bid].Value)
    iInitialBid = IIf(IsNull([Forms]![frm]![txt_initial_bid].Value), 0, [Forms]![frm]![txt_initial_bid].Value)
    iMinPosSlot = IIf(IsNull([Forms]![frm]![txt_min_pos_slot].Value), 0, [Forms]![frm]![txt_min_pos_slot].Value)
    iOpeningBid = IIf(IsNull([Forms]![frm]![txt_initial_bid].Value), -88, [Forms]![frm]![txt_initial_bid].Value)
    iWinningBid = IIf(IsNull([Forms]![frm]![txt_bid].Value), -99, [Forms]![frm]![txt_bid].Value)
    iMaxBid = IIf(IsNull([Forms]![frm]![txt_max_bid].Value), 0, [Forms]![frm]![txt_max_bid].Value)
    iPlayerId = Forms!frm!cbo_player.Value
    strBidder = Forms!frm!cbo_bidders.Value
    strNominator = IIf(IsNull(Forms!frm!txt_nominator.Value), "", Forms!frm!txt_nominator.Value)
    If iOpeningBid = iWinningBid Then
        If iMaxBid > 1 Then
            DoCmd.OpenForm "frm_cricket"
        End If
    End If

    strNominator = IIf(IsNull(Forms!frm_add_missing_player!cbo_nominator.Value), "", Forms!frm_add_missing_player!cbo_nominator.Value)

    fInitialBid = IIf(IsNull(Forms!frm_add_missing_player!txt_initial_bid.Value), 0, Forms!frm_add_missing_player!txt_initial_bid.Value)
    strNL = IIf(IsNull(Forms!frm_add_missing_player!cbo_NL.Value), "", Forms!frm_add_missing_player!cbo_NL.Value)

    strLastName = IIf(IsNull(Forms!frm_add_missing_player!txt_last_name.Value), "", Forms!frm_add_missing_player!txt_last_name.Value)
    strFullName = Trim(strFirstName) & " " & Trim(strLastName)
    Forms!frm_add_missing_player!txt_full_name.Value = strFullName
    
    If strOwner <> "" And strFirstName <> "" And strLastName <> "" And strNL <> "" And strContract <> "" And strPos <> "" Then
        If strOwner = "FA" Then
            iReadyToAddFlag = 1
        Else
            If strNominator <> "" And strPos <> "" And fSalary <> 0# And fInitialBid <> 0# And fInitialBid <= fSalary Then
                If strNominator = strOwner Then
                    iReadyToAddFlag = 1
                Else
                    If fSalary > fInitialBid Then

                    End If
                End If
            End If
        End If

        Forms!frm_add_missing_player!btn_add_player.SetFocus
    End If

Public Function fun_fix_autonumbers()
    Dim db As DAO.Database
    Dim tdf As TableDef
    Dim fld As DAO.Field
    Dim strTableName As String
    
    Set db = CurrentDb
    
    For Each tdf In db.TableDefs
        strTableName = tdf.Name
        If strTableName = "HR_points" Or strTableName = "RBI_points" Or strTableName = "SB_points" Or strTableName = "AVG_points" Or strTableName = "W_points" Or strTableName = "SV_points" Or strTableName = "ERA_points" Or strTableName = "WHIP_points" Then
            db.Execute "drop table " & strTableName
        End If
    Next
    db.Execute "CREATE TABLE HR_points (HR_rank COUNTER(1, 1) PRIMARY KEY, Owner VARCHAR(255), HR int)"
    db.Execute "CREATE TABLE RBI_points (RBI_rank COUNTER(1, 1) PRIMARY KEY, Owner VARCHAR(255), RBI int)"
    db.Execute "CREATE TABLE SB_points (SB_rank COUNTER(1, 1) PRIMARY KEY, Owner VARCHAR(255), SB int)"
    db.Execute "CREATE TABLE AVG_points (AVG_rank COUNTER(1, 1) PRIMARY KEY, Owner VARCHAR(255), AVG double)"
    db.Execute "CREATE TABLE W_points (W_rank COUNTER(1, 1) PRIMARY KEY, Owner VARCHAR(255), W int)"

Public Sub sub_in_draft_value_recalculation()
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Dim strStatSource As String
    Dim iFanGraphs As Long
    Dim iEnabled As Long
    Dim strHitterButton As String
    Dim strPitcherButton As String
    Dim ctlLoop As Control
    Dim strControlName As String
    Set db = CurrentDb
    Set rst = db.OpenRecordset("qry_external_stat_sources")
    Do While Not rst.EOF
        strStatSource = IIf(IsNull(rst!stat_source), "", rst!stat_source)
        iFanGraphs = IIf(IsNull(rst!fangraphs), 0, Abs(CLng(rst!fangraphs)))
        iEnabled = IIf(IsNull(rst!Enabled), 0, Abs(CLng(rst!Enabled)))
        strHitterButton = IIf(IsNull(rst!btn_name_import_hitters), "", rst!btn_name_import_hitters)
        strPitcherButton = IIf(IsNull(rst!btn_name_import_pitchers), "", rst!btn_name_import_pitchers)
        If Not CurrentProject.AllForms("frm_imports").IsLoaded Then
            DoCmd.OpenForm "frm_imports"
        End If
        For Each ctlLoop In Forms!frm_imports.Controls
            strControlName = ctlLoop.Name
            If strControlName = strHitterButton Or strControlName = strPitcherButton Then
                If iEnabled = 0 Then
                    ctlLoop.Enabled = False
                Else
                    ctlLoop.Enabled = True
                    ctlLoop.BackColor = CLR_BLUE_QUEUED
                End If
            End If
        Next
        rst.MoveNext
    Loop
    rst.Close
    Set rst = Nothing
    db.Close
    Set db = Nothing
        
    Forms!frm_imports.Requery
    Forms!frm_imports.Repaint
    
    Set db = CurrentDb
    Set rst = db.OpenRecordset("qry_external_stat_sources")
    Do While Not rst.EOF
        strStatSource = IIf(IsNull(rst!stat_source), "", rst!stat_source)
        iFanGraphs = IIf(IsNull(rst!fangraphs), 0, Abs(CLng(rst!fangraphs)))
        iEnabled = IIf(IsNull(rst!Enabled), 0, Abs(CLng(rst!Enabled)))
        strHitterButton = IIf(IsNull(rst!btn_name_import_hitters), "", rst!btn_name_import_hitters)
        strPitcherButton = IIf(IsNull(rst!btn_name_import_pitchers), "", rst!btn_name_import_pitchers)
        If iFanGraphs <> 0 And iEnabled <> 0 Then
            If strHitterButton <> "" Then
                Call fun_import_fangraphs_hitters(strStatSource, True)
            End If
            If strPitcherButton <> "" Then
                Call fun_import_fangraphs_pitchers(strStatSource, True)
            End If
            Call fun_set_sources
            Call fun_load_frm
        End If
        rst.MoveNext
    Loop
    rst.Close
    Set rst = Nothing
    db.Close
    Set db = Nothing
    
    If Not CurrentProject.AllForms("frm").IsLoaded Then
        DoCmd.OpenForm "frm"
    End If
    If CurrentProject.AllForms("frm_imports").IsLoaded Then
        DoCmd.Close acForm, "frm_imports"
    End If
    Forms!frm!cbo_player.Requery
    Forms!frm.SetFocus
    Forms!frm!cbo_player.SetFocus
End Sub

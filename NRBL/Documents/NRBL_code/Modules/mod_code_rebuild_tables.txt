Public Sub sub_rebuild_and_relink_table(strSQL As String, strTableName As String)
    Dim boolIsMock As Boolean
    Dim strDataPath As String
    Dim strFileNameWithPath As String
    boolIsMock = fn_db_is_mock()
    strDataPath = fn_get_db_data_path()
    If boolIsMock = True Then
        strFileNameWithPath = Replace(strDataPath & "\" & DB_DATA_MOCK, "\\", "\")
    Else
        strFileNameWithPath = Replace(strDataPath & "\" & DB_DATA, "\\", "\")
    End If
    Call sub_drop_table(strTableName)
    Call sub_build_table_from_sql(strSQL, strFileNameWithPath)
    Call link_access_table(strTableName, strTableName, strFileNameWithPath)
    set_status_bar ("")
End Sub
Public Sub sub_drop_table(strTableName As String)
    Dim strWorkTableName As String
    Dim db As DAO.Database
    Dim tdf As DAO.TableDef
    Dim strDataPath As String
    Dim strFrontFileNameWithPath As String
    Dim strBackFileNameWithPath As String
    Dim strSQL As String
    Dim boolIsMock As Boolean
    boolIsMock = fn_db_is_mock()
    strDataPath = fn_get_db_data_path()
    strFrontFileNameWithPath = Replace(fn_get_this_db_path() & "\" & DB_FRONT, "\\", "\")
    If boolIsMock = True Then
        strBackFileNameWithPath = Replace(strDataPath & "\" & DB_DATA_MOCK, "\\", "\")
    Else
        strBackFileNameWithPath = Replace(strDataPath & "\" & DB_DATA, "\\", "\")
    End If
    Set db = OpenDatabase(strBackFileNameWithPath)
    If CurrentProject.AllForms("frm").IsLoaded Then
        Forms!frm!frm_sub_status.Form.RecordSource = "tbl_subform_recordsource"
    End If
    For Each tdf In db.TableDefs
        strWorkTableName = tdf.Name
        If strWorkTableName = strTableName Then
            strSQL = "drop table " & strWorkTableName
            db.Execute strSQL, dbFailOnError
        End If
    Next tdf
    Set tdf = Nothing
    db.Close
    Set db = Nothing
    Set db = OpenDatabase(strFrontFileNameWithPath)
    For Each tdf In db.TableDefs
        strWorkTableName = tdf.Name
        If strWorkTableName = strTableName Then
            strSQL = "drop table " & strWorkTableName
            db.Execute strSQL, dbFailOnError
        End If
    Next tdf
    db.Close
    Set tdf = Nothing
    Set db = Nothing
End Sub
Public Sub sub_drop_local_table(strTableName As String)
    Dim db As DAO.Database
    Dim tdf As DAO.TableDef
    Dim strSQL As String
    Set db = CurrentDb
    For Each tdf In db.TableDefs
        If tdf.Name = strTableName Then
            strSQL = "drop table " & strTableName
            db.Execute strSQL, dbFailOnError
        End If
    Next tdf
    Set tdf = Nothing
    db.TableDefs.Refresh
    db.Close
    Set db = Nothing
End Sub
Public Sub sub_build_local_table_from_sql(strSQL As String)
    Dim db As DAO.Database
    Set db = CurrentDb
    db.Execute strSQL, dbFailOnError
    db.Close
    Set db = Nothing
End Sub
Public Sub sub_build_table_from_sql(strSQL As String, strFileNameWithPath As String)
    Dim db As DAO.Database
    Set db = OpenDatabase(strFileNameWithPath)
    db.Execute strSQL, dbFailOnError
    db.Close
    Set db = Nothing
End Sub
Public Function fn_rebuild_stat_points_table(strStat As String) As Boolean
    fn_rebuild_stat_points_table = False
    Dim strTableName As String
    Dim strSQL As String
    Dim strPositionCategory As String
    Dim boolIsCountingCategory As Boolean
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Dim boolFrmIsLoaded As Boolean
    strPositionCategory = ""
    boolFrmIsLoaded = CurrentProject.AllForms("frm").IsLoaded
    'If boolFrmIsLoaded = True Then
    '    DoCmd.Close acForm, "frm"
    'End If
    strTableName = Trim(strStat) & "_points"
    Set db = CurrentDb
    strSQL = "select * from stat_categories where is_active = True and stat_category_name = " & """" & Replace(strStat, """", """""") & """" & ";"
    Set rst = db.OpenRecordset(strSQL)
    If Not rst.EOF Then
        strPositionCategory = IIf(IsNull(rst!position_category), "", rst!position_category)
        boolIsCountingCategory = rst!is_counting_category
    End If
    rst.Close
    Set rst = Nothing
    db.Close
    Set db = Nothing
    If strPositionCategory <> "" Then
        Call sub_drop_local_table(strTableName)
        strSQL = "create table [" & Replace(strTableName, """", """""") & "] ( [" & Replace(strStat, """", """""") & "_rank] counter(1,1) primary key,"
        strSQL = strSQL & " Owner varchar, [" & Replace(strStat, """", """""") & "] "
        If boolIsCountingCategory Then
            strSQL = strSQL & " long )"
        Else
            strSQL = strSQL & " double )"
        End If
        Call sub_build_local_table_from_sql(strSQL)
        'Call sub_rebuild_and_relink_table(strSQL, strTableName)
        'If boolFrmIsLoaded = True Then
        '    DoCmd.OpenForm "frm"
        'End If
        fn_rebuild_stat_points_table = True
        Exit Function
    Else
        MsgBox "Error obtaining stat info for strStat = " & IIf(IsNull(strStat), "", strStat)
    End If
End Function

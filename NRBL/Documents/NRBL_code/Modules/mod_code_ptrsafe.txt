Public Sub search_querydef_sql()
    Dim dbFront As DAO.Database
    Dim qdf As DAO.QueryDef
    Dim queryName As String
    Dim querySQL As String
    Set dbFront = CurrentDb()
    For Each qdf In dbFront.QueryDefs
        queryName = qdf.Name
        querySQL = qdf.SQL
        If InStr(1, querySQL, "AB") > 0 Then
            MsgBox queryName & " | " & querySQL
        End If
    Next qdf
    MsgBox "Done."
End Sub

    Set dbFront = CurrentDb

        strRemoteTableName = rst!table_name

Public Function fn_db_is_mock() As Boolean
    fn_db_is_mock = False
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Set db = CurrentDb
    Set rst = db.OpenRecordset("mock_selector")
    If Not rst.EOF Then
        fn_db_is_mock = IIf(IsNull(rst!is_mock), False, rst!is_mock)
    End If
    db.Close
    Set db = Nothing
End Function
Public Function fun_set_sources()
    Dim db As DAO.Database
    Set db = CurrentDb
    
    Dim rst As DAO.Recordset
    Dim strRS As String
    Dim strDBFile As String
    
    Dim strPitcherSource As String
    Dim strHitterSource As String
    Dim strSrc As String
    
    Set rst = db.OpenRecordset("qry_position_category_sources_hitter")
    strHitterSource = IIf(IsNull(rst!src), "", rst!src)
    rst.Close
    Set rst = db.OpenRecordset("qry_position_category_sources_pitcher")
    strPitcherSource = IIf(IsNull(rst!src), "", rst!src)
    rst.Close
    Set rst = Nothing
    db.Close
    Set db = Nothing
    strSrc = "Hitter Source:  " & strHitterSource & "          Pitcher Source:  " & strPitcherSource
    If CurrentProject.AllForms("frm").IsLoaded Then
        Forms!frm!lab_sources.Caption = strSrc
        Forms!frm.Requery
        Forms!frm.Repaint
    End If
End Function
Public Function fun_load_frm()
    DoCmd.SetWarnings True
    Dim strNow As Date
    Dim iYear As Long
    Dim strYear As String
    Dim strHeader As String
    Dim strCaption As String
    Dim strFooter As String
    Dim iIPFloor As Long
    Dim iABFloor As Long
    Dim fInSeasonCap As Currency
    Dim fDraftCap As Currency
    Dim fFAAB As Currency
    Dim iTeamCt As Integer
    Dim iHitterSlots As Integer
    Dim iPitcherSlots As Integer
    Dim iSwingmanSlots As Integer
    Dim dtDraftDay As Date
    
    
    strNow = Now()
    
    Dim db As DAO.Database
    Set db = CurrentDb
    
    Dim rst As DAO.Recordset
    Dim strRS As String
    Dim strDBFile As String
    
    Call fun_set_sources
        
    Set rst = db.OpenRecordset("NRBL_application_settings")
    iIPFloor = rst!IP_floor
    iABFloor = rst!AB_floor
    iYear = rst!historical_year
    iTeamCt = rst!active_team_count
    fInSeasonCap = rst!in_season_cap
    fDraftCap = rst!draft_cap
    fFAAB = rst!FAAB
    iHitterSlots = rst!hitter_slots
    iPitcherSlots = rst!pitcher_slots
    iSwingmanSlots = rst!swingman_slots
    dtDraftDay = rst!draft_date
    rst.Close
    Set rst = Nothing
    If CurrentProject.AllForms("frm").IsLoaded Then
        Forms!frm!txt_IP_floor.Value = iIPFloor
        Forms!frm!txt_AB_floor.Value = iABFloor
    End If
    strYear = CStr(iYear)
    Dim strDBName As String
    strDBFile = db.Name
    strHeader = strYear & " NRBL Auction"
    strCaption = strHeader & " (NL-Only, " & iTeamCt & " Teams) | " & fn_get_stat_position_category_ct_description & " | Draft Cap: $" & CStr(CLng(fDraftCap)) & " | In-Season Cap: $" & CStr(CLng(fInSeasonCap)) & " | FAAB $" & CStr(CLng(fFAAB))
    strCaption = strCaption & " | Slots - H: " & iHitterSlots & "  P: " & iPitcherSlots & "  SW: " & iSwingmanSlots
    strFooter = strCaption & " | Floors - IP: " & iIPFloor & "  AB: " & iABFloor
    If fn_db_is_mock = True Then
        strHeader = strHeader & " *** MOCK ***"
        If CurrentProject.AllForms("frm").IsLoaded Then
            Forms!frm!lab_merry_christmas.Visible = False
            Forms!frm!baseball_christmas.Visible = False
            If dtDraftDay < strNow Then
                Forms!frm!lab_merry_christmas.Visible = True
                Forms!frm!baseball_christmas.Visible = True
            End If
            Forms!frm!btn_execute_cuts.Enabled = True
            Forms!frm!btn_reset_to_pre_draft.Enabled = True
        End If
    Else
        If CurrentProject.AllForms("frm").IsLoaded Then
            Forms!frm!lab_merry_christmas.Visible = False
            Forms!frm!baseball_christmas.Visible = False
            If dtDraftDay < strNow Then
                Forms!frm!lab_merry_christmas.Visible = True
                Forms!frm!baseball_christmas.Visible = True
            End If
            Forms!frm!btn_execute_cuts.Enabled = False
            Forms!frm!btn_reset_to_pre_draft.Enabled = False
        End If
    End If
    DoCmd.OpenQuery "qry_team_position_slot_pre_draft_slot_summary_delete"
    DoCmd.OpenQuery "qry_team_position_slot_pre_draft_slot_summary_append"
    If CurrentProject.AllForms("frm").IsLoaded Then
        Forms!frm!lab_header.Caption = strHeader
        Forms!frm.Caption = strCaption
        Forms!frm!lab_facts.Caption = strFooter
        'Forms!frm!btn_reset_to_pre_draft.Enabled = True
        'If strNow > "4/8/2017 06:30" Then  ' If strNow > "3/29/2014 10:30" Then
        '    Forms!frm!btn_reset_to_pre_draft.Enabled = False
        'End If
    
        Set rst = db.OpenRecordset("tbl_subform_recordsource")
        strRS = rst!rs
        rst.Close
        Set rst = Nothing
        'Forms!frm!frm_sub_status.Form.RecordSource = strRS
        Forms!frm!frm_sub_status.Requery
        Dim iDraftSlots As Long
        Set rst = db.OpenRecordset("tbl_team_position_slot_pre_draft_slot_summary")
        iDraftSlots = rst!non_keeper_slot_ct
        rst.Close
        Set rst = Nothing
        Forms!frm!txt_pre_draft_slot_ct.Value = iDraftSlots
        Call fun_get_next_drafter
        Dim curInitialBid As Currency
        Dim iMaxBidCt As Long
        Dim curMaxBid As Double
        Dim strOwner As String
        Dim strWorkOwner As String
        Dim strOwnerTeam As String
        Dim strWorkCaption As String
        strOwner = Forms!frm!txt_nominator.Value
        strOwnerTeam = ""
        Set rst = db.OpenRecordset("NRBL_owners")
        Do While Not rst.EOF
            strWorkOwner = IIf(IsNull(rst!owner_code), "", rst!owner_code)
            If strWorkOwner = strOwner Then
                strOwnerTeam = IIf(IsNull(rst!NRBL_team), "", rst!NRBL_team)
            End If
            rst.MoveNext
        Loop
        rst.Close
        Set rst = Nothing
        
        If CurrentProject.AllForms("frm").IsLoaded Then
            strWorkCaption = Forms!frm!lab_draft_ct.Caption & " - " & strOwnerTeam & "'"
            If Right(strOwnerTeam, 1) <> "s" Then
                strWorkCaption = strWorkCaption & "s"
            End If
            strWorkCaption = strWorkCaption & " nomination"
            Forms!frm!lab_draft_ct.Caption = strWorkCaption
        End If
    
        curInitialBid = Forms!frm!txt_initial_bid.Value
        DoCmd.OpenQuery "qry_max_bid_delete"
        DoCmd.OpenQuery "qry_nominator_max_bid_append"
        Set rst = db.OpenRecordset("qry_max_bid_ct")
        iMaxBidCt = rst!ct
        rst.Close
        Set rst = Nothing
        If iMaxBidCt = 0 Then
            curMaxBid = 0
        Else
            Set rst = db.OpenRecordset("tbl_max_bid")
            curMaxBid = rst!max_bid_amt
            rst.Close
            Set rst = Nothing
        End If
'''''
' curMaxBid = 99999.99
'''''
        Forms!frm!txt_max_bid.Value = curMaxBid
        Call fun_set_status
        If curInitialBid > curMaxBid Then
            If curMaxBid > 0 Then
                Forms!frm!btn_draft.Enabled = False
                DoCmd.Beep
                MsgBox "The nominator (" & Trim(strOwner) & ") has exceeded their maximum bid of " & FormatCurrency(curMaxBid, 0) & " in the initial bid."
            End If
        End If
        Forms!frm!cbo_player.SetFocus
        'Forms!frm!btn_selected_player_projections.Enabled = False
        Forms!frm!txt_bid.Value = curInitialBid
        Call fun_load_stat_copy_aggs
    End If
End Function
Public Function fun_set_status()
    Dim strOwner As String
    Dim strStatus As String
    Dim strPosNeeded As String
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Dim iPsNeeded As Integer
    Dim iOFsNeeded As Integer
    Dim iCNeeded As Integer
    Dim i1BNeeded As Integer
    Dim i2BNeeded As Integer
    Dim iSSNeeded As Integer
    Dim i3BNeeded As Integer
    Dim iMINeeded As Integer
    Dim iCMNeeded As Integer
    Dim iCOFNeeded As Integer
    Dim iUTNeeded As Integer
    Dim iSWNeeded As Integer
    Dim iPlayersNeeded As Integer
    Dim iNeedsComma As Integer
    Dim iMaxBid As Integer
    Dim iDraftPickCt As Integer
    Dim strLastName As String
    Dim strFirstName As String
    Dim strOpeningBidder As String
    Dim strWinningBidder As String
    Dim iOpeningBid As Integer
    Dim iSal As Integer
    Dim strNL As String
    Dim strPos As String
    Dim strLastDraftPick As String
    Dim iCt As Integer
    strLastDraftPick = ""
    Set db = CurrentDb()
    strOwner = IIf(IsNull(Forms![frm]!txt_nominator.Value), "", Forms![frm]!txt_nominator.Value)
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_delete"
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_append"
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_ct_delete"
    DoCmd.OpenQuery "qry_team_needs_summary_nominator_ct_append"
    Set rst = db.OpenRecordset("tbl_team_needs_summary_nominator_ct")
    iCt = rst!ct
    rst.Close
    Set rst = Nothing
    iPsNeeded = 0
    iOFsNeeded = 0
    iCNeeded = 0
    i1BNeeded = 0
    i2BNeeded = 0
    iSSNeeded = 0
    i3BNeeded = 0
    iMINeeded = 0
    iCMNeeded = 0
    iCOFNeeded = 0
    iUTNeeded = 0
    iSWNeeded = 0
    iPlayersNeeded = 0
    iNeedsComma = 0
    If iCt = 0 Then
        strStatus = "The auction draft has completed."
    End If
    If iCt > 0 Then
        Set rst = db.OpenRecordset("tbl_team_needs_summary_nominator")
        iPsNeeded = IIf(IsNull(rst!Ps_needed), 0, rst!Ps_needed)
        iOFsNeeded = IIf(IsNull(rst!OFs_needed), 0, rst!OFs_needed)
        iCNeeded = IIf(IsNull(rst!needs_C), 0, rst!needs_C)
        i1BNeeded = IIf(IsNull(rst!needs_1B), 0, rst!needs_1B)
        i2BNeeded = IIf(IsNull(rst!needs_2B), 0, rst!needs_2B)
        iSSNeeded = IIf(IsNull(rst!needs_SS), 0, rst!needs_SS)
        i3BNeeded = IIf(IsNull(rst!needs_3B), 0, rst!needs_3B)
        iMINeeded = IIf(IsNull(rst!needs_MI), 0, rst!needs_MI)
        iCMNeeded = IIf(IsNull(rst!needs_CM), 0, rst!needs_CM)
        iCOFNeeded = IIf(IsNull(rst!needs_COF), 0, rst!needs_COF)
        iUTNeeded = IIf(IsNull(rst!needs_UT), 0, rst!needs_UT)
        iSWNeeded = IIf(IsNull(rst!needs_SW), 0, rst!needs_SW)
        iPlayersNeeded = IIf(IsNull(rst!players_needed), 0, rst!players_needed)
        iNeedsComma = 0
        rst.Close
        Set rst = db.OpenRecordset("tbl_max_bid")
        iMaxBid = IIf(IsNull(Forms!frm!txt_max_bid.Value), 0, Forms!frm!txt_max_bid.Value)
        If iMaxBid = 0 Then
            iMaxBid = IIf(IsNull(rst!max_bid_amt), 0, rst!max_bid_amt)
        End If
        rst.Close
        Set rst = Nothing
        DoCmd.OpenQuery "qry_draft_pick_ct_delete"
        DoCmd.OpenQuery "qry_draft_pick_ct_append"
    End If
    DoCmd.OpenQuery "qry_most_recent_drafted_player_delete"
    DoCmd.OpenQuery "qry_most_recent_drafted_player_append"
    Set rst = db.OpenRecordset("tbl_draft_pick_ct")
    iDraftPickCt = rst!draft_pick_ct
    rst.Close
    Set rst = Nothing
    If iDraftPickCt > 0 Then
        Set rst = db.OpenRecordset("tbl_most_recent_drafted_player")
        strLastName = rst!last_name
        strFirstName = rst!first_name
        strOpeningBidder = rst!opening_bidder
        strWinningBidder = rst!Owner
        iOpeningBid = rst!opening_bid
        iSal = rst!Sal
        strNL = rst!NL
        strPos = rst!pos
        rst.Close
        Set rst = Nothing
        strLastDraftPick = "Last drafted: " & Trim(strFirstName) & " " & Trim(strLastName)
        strLastDraftPick = strLastDraftPick & " (" & Trim(strNL) & " - " & Trim(strPos) & ")"
        strLastDraftPick = strLastDraftPick & " by " & strWinningBidder
        strLastDraftPick = strLastDraftPick & "     Contract: Y1 $" & iSal
        If iSal = iOpeningBid Then
            strLastDraftPick = strLastDraftPick & "     *** Crickets ***"
        Else
            strLastDraftPick = strLastDraftPick & "     [Nom:  " & strOpeningBidder & " $" & iOpeningBid & "]"
        End If
    End If
    Set rst = Nothing
    db.Close
    Set db = Nothing
    If iPlayersNeeded = 0 Then
        strStatus = "The auction has ended."
    End If
    If iPlayersNeeded > 0 Then
        strStatus = strOwner & " needs: "
        If iPsNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            If iPsNeeded > 1 Then
                strStatus = strStatus & iPsNeeded & " P"
            Else
                strStatus = strStatus & "P"
            End If
            iNeedsComma = 1
        End If
        If iOFsNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            If iOFsNeeded > 1 Then
                strStatus = strStatus & iOFsNeeded & " OF"
            Else
                strStatus = strStatus & "OF"
            End If
            iNeedsComma = 1
        End If
        If iCNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "C"
            iNeedsComma = 1
        End If
        If i1BNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "1B"
            iNeedsComma = 1
        End If
        If i2BNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "2B"
            iNeedsComma = 1
        End If
        If iSSNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "SS"
            iNeedsComma = 1
        End If
        If i3BNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "3B"
            iNeedsComma = 1
        End If
        If iMINeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "MI"
            iNeedsComma = 1
        End If
        If iCMNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "CM"
            iNeedsComma = 1
        End If
        If iCOFNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "C/OF"
            iNeedsComma = 1
        End If
        If iUTNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "UT"
            iNeedsComma = 1
        End If
        If iSWNeeded > 0 Then
            If iNeedsComma = 1 Then
                strStatus = strStatus & ", "
            End If
            strStatus = strStatus & "SW"
            iNeedsComma = 1
        End If
        If iMaxBid = 1 Then
            strStatus = strStatus & "       *** DOLLAR DAZE ***"
        Else
            strStatus = strStatus & "       Max Bid:  $" & iMaxBid
        End If
        
    End If
    strStatus = strStatus & vbCrLf
    If iDraftPickCt = 0 Then
        strStatus = strStatus & "There have not yet been any players drafted.  You can almost taste the optimism!"
    Else
        strStatus = strStatus & strLastDraftPick
    End If
    Forms![frm]!lab_status_bar.Caption = strStatus
    Forms![frm].Repaint
End Function

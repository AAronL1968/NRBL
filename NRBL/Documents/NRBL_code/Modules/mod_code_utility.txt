Option Compare Database
Option Explicit
Public Sub set_status_bar(strStatus As String)
    Dim varStatus As Variant
    If IIf(IsNull(strStatus), "", strStatus) = "" Then
        varStatus = SysCmd(acSysCmdClearStatus)
    Else
        varStatus = SysCmd(acSysCmdSetStatus, strStatus)
    End If
End Sub
Public Function fn_get_rowcount(strRecordsetName As String) As Long
    Dim iCt As Long
    Dim db As DAO.Database
    Dim rst As DAO.Recordset
    Dim strSQL As String
    DoCmd.OpenQuery "qry_loc_cts_delete"
    strSQL = "insert into [__cts] ( ct ) select count(*) as ct from [" & Trim(strRecordsetName) & "];"
    Set db = CurrentDb
    db.Execute strSQL, dbFailOnError
    iCt = CLng(0)
    Set rst = db.OpenRecordset("__cts")
    If Not rst.EOF Then
        iCt = IIf(IsNull(rst!ct), CLng(0), CLng(rst!ct))
    End If
    rst.Close
    Set rst = Nothing
    db.Close
    Set db = Nothing
    fn_get_rowcount = iCt
End Function
Public Sub sub_delete_all_rows(strTableName As String)
    Dim db As DAO.Database
    Dim strSQL As String
    strSQL = "delete * from [" & strTableName & "]"
    Set db = CurrentDb
    db.Execute strSQL, dbFailOnError
    db.Close
    Set db = Nothing
End Sub
Public Function fn_get_interval_description(dtStart As Date, dtEnd As Date) As String
    If dtEnd < dtStart Then
        fn_get_interval_description = "Error - end date cannot be earlier the start date."
        Exit Function
    End If
    If dtEnd = dtStart Then
        fn_get_interval_description = "Sub-second"
        Exit Function
    End If
    Dim iSeconds As Long
    Dim iMinutes As Long
    Dim iRemainingSeconds As Long
    Dim strInterval As String
    iSeconds = DateDiff("s", dtStart, dtEnd)
    iMinutes = Int(iSeconds / 60)
    iRemainingSeconds = iSeconds Mod 60
    strInterval = iMinutes & " minute"
    If iMinutes <> 1 Then
        strInterval = strInterval & "s"
    End If
    strInterval = strInterval & " and " & iRemainingSeconds & " second"
    If iSeconds <> 1 Then
        strInterval = strInterval & "s"
    End If
    fn_get_interval_description = strInterval
End Function
Public Function fn_copy_file(strFromLocation As String, strToLocation As String) As Boolean
    fn_copy_file = False
    Dim fso As Object
    Dim strDirResult As String
    Dim boolFolderExists As Boolean
    Dim iLastBackslashPos As Integer
    Dim strTargetDir As String
    Set fso = VBA.CreateObject("Scripting.FileSystemObject")
    strDirResult = Dir(strFromLocation)
    If strDirResult = "" Or IsNull(strDirResult) = True Then
        MsgBox "The specified source file does not exist." & vbCrLf & " " & vbCrLf & strFromLocation
        Exit Function
    End If
    iLastBackslashPos = InStrRev(strToLocation, "\")
    strTargetDir = Left(strToLocation, iLastBackslashPos)
    boolFolderExists = fso.FolderExists(strTargetDir)
    If boolFolderExists = True Then
        fn_copy_file = True
        Call fso.CopyFile(strFromLocation, strToLocation, True)
        Exit Function
    Else
        MsgBox "The specified destination folder does not exist." & vbCrLf & " " & vbCrLf & strTargetDir
        Exit Function
    End If
End Function
Public Function fn_get_string_datestamp() As String
    Dim strMonth As String
    Dim strDay As String
    Dim strYear As String
    Dim iMonth As Integer
    Dim iDay As Integer
    Dim iYear As Integer
    Dim dtNow As Date
    Dim strDateStamp As String
    dtNow = Now()
    iYear = DatePart("yyyy", dtNow)
    iMonth = DatePart("m", dtNow)
    iDay = DatePart("d", dtNow)
    strYear = CStr(iYear)
    strMonth = Right(CStr(100 + iMonth), 2)
    strDay = Right(CStr(100 + iDay), 2)
    strDateStamp = strYear & strMonth & strDay
    fn_get_string_datestamp = strDateStamp
End Function
Public Function fn_get_date_as_string(ByVal dtDate As Date) As String
    Dim strMonth As String
    Dim strDay As String
    Dim strYear As String
    Dim iMonth As Integer
    Dim iDay As Integer
    Dim iYear As Integer
    Dim dtNow As Date
    Dim strDate As String
    iYear = DatePart("yyyy", dtDate)
    iMonth = DatePart("m", dtDate)
    iDay = DatePart("d", dtDate)
    strYear = CStr(iYear)
    strMonth = Right(CStr(100 + iMonth), 2)
    strDay = Right(CStr(100 + iDay), 2)
    strDate = strYear & "-" & strMonth & "-" & strDay
    fn_get_date_as_string = strDate
End Function

Public Sub sub_table_rowct_study()
    Call sub_drop_table("___table_rowct_study")
    Dim db As DAO.Database
    Dim tdf As DAO.TableDef
    Dim strSQL As String
    Dim strFrontFileNameWithPath As String
    strFrontFileNameWithPath = Replace(fn_get_this_db_path() & "\" & DB_FRONT, "\\", "\")
    strSQL = "create table ___table_rowct_study ( table_name varchar primary key, row_ct long );"
    Call sub_build_table_from_sql(strSQL, strFrontFileNameWithPath)
    Set db = CurrentDb
    For Each tdf In db.TableDefs
        strSQL = "insert into [___table_rowct_study] ( table_name, row_ct )"
        strSQL = strSQL & " select " & """" & tdf.Name & """" & " as table_name, count(*) as row_ct from [" & tdf.Name & "];"
        db.Execute strSQL, dbFailOnError
    Next tdf
    Set tdf = Nothing
    db.Close
    Set db = Nothing
End Sub

Public Sub sub_rebuild_REGAL_MainDataSet()
    Dim strSQL As String
    Dim boolIsMock As Boolean
    Dim strDataPath As String
    Dim strFileNameWithPath As String
    Dim strTableName As String
    strTableName = "REGAL_MainDataSet"
    boolIsMock = fn_db_is_mock()
    strDataPath = fn_get_db_data_path()

        strFileNameWithPath = Replace(strDataPath & "\" & DB_DATA_MOCK, "\\", "\")
    Else
        strFileNameWithPath = Replace(strDataPath & "\" & DB_DATA, "\\", "\")
    End If
    Call sub_drop_table(strTableName)
    strSQL = "create table REGAL_MainDataSet ( ID counter(1,1) primary key,"
    strSQL = strSQL & " FilmKey varchar, LocationCode varchar, BusinessDate varchar, TicketClass varchar,"
    strSQL = strSQL & " TicketType_Short varchar, TicketType varchar, ShowtimeLocal datetime, Auditorium varchar,"
    strSQL = strSQL & " PrintId long, Attendance long, QtyVoid long, QtyRefund long, QtySwap long, BoxGross currency, BoxNet currency )"
    Call sub_build_table_from_sql(strSQL, strFileNameWithPath)
    Call link_access_table(strTableName, strTableName, strFileNameWithPath)
    set_status_bar ("")
End Sub
Public Sub sub_check_AllModules_for_OptionExplicit()
    Dim objProject As Access.CurrentProject
    Dim objModule As AccessObject
    Dim modWork As Access.Module
    Dim strModuleName As String
    Dim strModuleText As String
    Dim strTargetText As String
    Dim iModules As Long
    Dim iLines As Long
    Dim boolHasOptionExplicit As Boolean
    strTargetText = "Option Explicit"
    boolHasOptionExplicit = False
    Set objProject = Application.CurrentProject
    For Each objModule In objProject.AllModules
        strModuleName = objModule.Name
        DoCmd.OpenModule strModuleName
        Set modWork = Modules(strModuleName)
        iLines = modWork.CountOfLines
        strModuleText = modWork.Lines(1, iLines)
        Set modWork = Nothing
        If InStr(strModuleText, strTargetText) > 0 Then
            boolHasOptionExplicit = True
        Else
            boolHasOptionExplicit = False
        End If
        If boolHasOptionExplicit = False Then
            Debug.Print "  CurrentProject.AllModules - " & CStr(boolHasOptionExplicit) & " - " & strModuleName
        End If
        'Debug.Print "  CurrentProject.AllModules - " & CStr(boolHasOptionExplicit) & " - " & strModuleName
    Next objModule
    Set modWork = Nothing
    Set objModule = Nothing
    Set objProject = Nothing
End Sub

Public Sub sub_find_string_in_query(strTextToFind As String)
    Dim db As DAO.Database
    Dim qdf As DAO.QueryDef
    Dim strSQL As String
    Dim strQueryName As String
    Dim iCtr As Long
    iCtr = 0
    Set db = CurrentDb
    For Each qdf In db.QueryDefs
        strQueryName = qdf.Name
        strSQL = qdf.SQL
        If InStr(strSQL, strTextToFind) > 0 Then
            iCtr = iCtr + 1
            Debug.Print strQueryName & " ' " & iCtr ' & vbCrLf
        End If
    Next qdf
    db.Close
    Set db = Nothing
End Sub
